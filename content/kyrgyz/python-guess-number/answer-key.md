## 'Сандарды тап' оюнун түзүү

```python
import random

# Бул жерден random.randint() функциясын колдонуп сан түзүңүз
# Ошондой эле бул жерден божомолдоого уруксат берилген оппонеттердин санын белгилеңиз
randomNumber = random.randint(1,10)
guessesLeft = 3

# Бул жерде оюнчуга кайталап сан сураган жана аны кокус сан менен салыштырган цикл түзүңүз
# Оюнчу чексиз аракет жасабашы үчүн, калган божомолдордун санын азайтууну унутпаңыз!
while guessesLeft > 0:
    print('\n1ден 10го чейинки санды болжоп көрүңүз:')
    
    # Бул жерде колдонуучудан маалымат сураңыз
    print('Сураныч, сандарды гана киргизиңиз')
    userinput =  int(input())

    # Жаңы кабыл алынган маалыматты randomNumber менен салыштырып, оюнчуга азайтуу же көбүрөөк божомолдоо керектигин айтып туруңуз
    # Эгер божомол туура болсо, циклден чыгуу үчүн break колдонгонду унутпаңыз
    if userinput == randomNumber:
      print("Туура таптыңыз! Куттуктайм :)! Кайра ойноо үчүн 'run' басыңыз")
      break
    elif userinput > randomNumber:
      print("Сиздин саныңыз өтө чоң. Аз санды божомдогонго аракет кылыңыз.")
    else:
      print("Сиздин саныңыз өтө кичине. Чоң санды божомдогонго аракет кылыңыз.")
    
    # Оюнчуга калган аракеттердин санын азайтыңыз
    guessesLeft = guessesLeft -1

# Оюнчунун бардык божомолдору түгөнүп калган-жоктугун текшериңиз. Эгерде бардык аракеттер түгөнсө, аларга оюнду уттурганы тууралуу билдирүү айтыңыз
if guessesLeft == 0:
  print("Сиздин аракеттериңиз түгөнүп калды. :/ Туура сан " + str(randomNumber) + " эле!") # ЖЕ ("Сиздин аракеттериңиз түгөнүп калды. :/ Туура сан ", randomNumber)
else:
  pass
```
